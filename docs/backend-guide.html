ขั้นตอนการติดตั้งและการรันระบบ (Setup & Migration Guide)

createdb innoaward_dev

# 2. ติดตั้ง Dependencies และ Generate Prisma Client
cd server
bun install
bunx prisma migrate dev --name init_innoaward_normalized_schema

bun test
